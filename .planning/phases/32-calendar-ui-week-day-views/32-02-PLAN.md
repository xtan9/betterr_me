2. **All-day row**: Render `<AllDayRow>` above the grid. Filter events where `start_time === null` for each date column.

3. **Time gutter** (left column, ~60px wide): 24 rows labeled with formatted hours. Format hours in 12h with AM/PM using the locale. Example labels: "12 AM", "1 AM", ..., "12 PM", "1 PM", ..., "11 PM". Place each label at `top: hour * 48px`, vertically centered on the grid line.

4. **Day columns**: CSS grid with `grid-template-columns: 60px repeat(N, 1fr)` where N = `dates.length`. Each column is relatively positioned for absolute event block placement.

5. **Hour rows**: 24 rows of `height: 48px` each (total 1152px). Full-width `border-b border-border` on each hour line. Half-hour dashed sub-grid: a lighter `border-b border-dashed border-border/50` line at the 24px midpoint of each 48px row (per D-02).

6. **Event rendering within columns**: For each date column, get timed events (where `start_time !== null`), run overlap detection, and render `<EventBlock>` components positioned absolutely.

7. **Overlap detection algorithm** (per D-03, Google Calendar style):
   ```ts
   function computeOverlapColumns(events: ExpandedCalendarEvent[]): Map<string, { column: number; totalColumns: number }> {
     // Sort by start_time, then by duration descending
     const sorted = [...events].sort((a, b) => {
       const cmp = (a.start_time || "").localeCompare(b.start_time || "");
       if (cmp !== 0) return cmp;
       return getDurationMinutes(b) - getDurationMinutes(a);
     });

     // For each event, find the first available column
     const columns: { eventId: string; column: number; startMin: number; endMin: number }[] = [];
     for (const event of sorted) {
       const startMin = timeToMinutes(event.start_time!);
       const endMin = event.end_time ? timeToMinutes(event.end_time) : startMin + 60;
       let col = 0;
       while (columns.some(c => c.column === col && c.startMin < endMin && c.endMin > startMin)) {
         col++;
       }
       columns.push({ eventId: event.id, column: col, startMin, endMin });
     }

     // Determine total columns per overlap group
     const result = new Map<string, { column: number; totalColumns: number }>();
     for (const entry of columns) {
       const overlapping = columns.filter(c =>
         c.startMin < entry.endMin && c.endMin > entry.startMin
       );
       const maxCol = Math.max(...overlapping.map(c => c.column)) + 1;
       result.set(entry.eventId, { column: entry.column, totalColumns: maxCol });
     }
     return result;
   }
   ```

8. **Helper functions** (defined inside the file):
   - `timeToMinutes(time: string): number` — converts "HH:MM:SS" or "HH:MM" to total minutes since midnight
   - `getDurationMinutes(event: ExpandedCalendarEvent): number` — returns event duration in minutes (default 60 if no end_time)
   - `formatHourLabel(hour: number): string` — returns formatted hour label like "12 AM", "1 PM"

9. **Current time indicator**: Render `<CurrentTimeIndicator>` inside today's column only (per D-11). Pass the column's relative position context.

10. **Click interaction on time slots** (per D-14, EVNT-07): Each half-hour cell is clickable. On click, compute the time from the click position:
    - `offsetY` = click Y relative to the scrollable grid container
    - `minutes = Math.floor(offsetY / 48 * 60)`
    - Snap to nearest 15-minute interval: `snapped = Math.round(minutes / 15) * 15`
    - Convert to HH:MM string and call `onTimeSlotClick(date, timeString)`

11. **Drag interaction** (per D-17, D-18, EVNT-08): Track mouse events on the grid:

    - Use state: `isDragging: boolean`, `dragStart: { date: Date; minutes: number } | null`, `dragEnd: { minutes: number } | null`
    - Visual feedback: absolute-positioned div with `bg-primary/20 rounded` covering the drag range

12. **Styling**: Use Tailwind classes. The grid background should be `bg-background`, grid lines `border-border`, time labels `text-xs text-muted-foreground`.

```tsx
import { memo } from "react";
import type { ExpandedCalendarEvent } from "@/lib/calendar/recurrence";

interface EventBlockProps {
  event: ExpandedCalendarEvent;
  /** Top offset in pixels from the grid top */
  top: number;
  /** Height in pixels (proportional to duration) */
  height: number;
  /** CSS left percentage (e.g., "0%", "50%") for overlap column positioning */
  left: string;
  /** CSS width percentage (e.g., "100%", "50%") for overlap column sizing */
  width: string;
  /** Callback when the block is clicked */
  onClick?: (event: ExpandedCalendarEvent) => void;
}

function formatTimeRange(start: string | null, end: string | null): string {
  if (!start) return "";
  const fmtStart = start.slice(0, 5); // HH:MM
  if (!end) return fmtStart;
  const fmtEnd = end.slice(0, 5);
  return `${fmtStart} – ${fmtEnd}`;
}

export const EventBlock = memo(function EventBlock({
  event,
  top,
  height,
  left,
  width,
  onClick,
}: EventBlockProps) {
  const hasCustomColor = !!event.color;
  const isShort = height < 30; // Less than ~37 minutes — compact layout

  const bgStyle = hasCustomColor
    ? { backgroundColor: `${event.color}20` }
    : {};
  const borderStyle = hasCustomColor && event.color
    ? { borderLeftColor: event.color }
    : {};

  return (
    <button
      type="button"
      className={`
        absolute rounded-md px-1.5 py-0.5
        border-l-2 overflow-hidden
        text-xs text-left cursor-pointer
        hover:opacity-80 transition-opacity
        ${hasCustomColor ? "" : "bg-[hsl(var(--calendar-event-muted))] border-l-[hsl(var(--calendar-event))]"}
      `}
      style={{
        top: `${top}px`,
        height: `${Math.max(height, 20)}px`, // minimum 20px height
        left,
        width,
        ...bgStyle,
        ...borderStyle,
      }}
      onClick={(e) => {
        e.stopPropagation();
        onClick?.(event);
      }}
      title={event.title}
    >
      {isShort ? (
        <span className="truncate block">{event.title}</span>
      ) : (
        <>
          <span className="font-medium truncate block">{event.title}</span>
          <span className="text-muted-foreground truncate block">
            {formatTimeRange(event.start_time, event.end_time)}
          </span>
        </>
      )}
    </button>
  );
});
```

- Uses `button` element for accessibility (clickable, keyboard-focusable)

- Short events (<30px height) show title only; taller events show title + time range
- Color logic mirrors `EventChip` pattern: default uses CSS variables, custom uses inline styles
- `e.stopPropagation()` on click prevents the time slot click handler from firing
- Positioned absolutely with `top`, `height`, `left`, `width` from parent overlap calculation

```tsx
"use client";

import { useState } from "react";
import { useTranslations } from "next-intl";
import { EventChip } from "./event-chip";
import type { ExpandedCalendarEvent } from "@/lib/calendar/recurrence";
import { getLocalDateString } from "@/lib/utils";

interface AllDayRowProps {
  /** Array of dates (columns) in the grid */
  dates: Date[];
  /** Map of date string -> events for that date */
  events: Map<string, ExpandedCalendarEvent[]>;
  /** Callback when an event is clicked */
  onEventClick?: (event: ExpandedCalendarEvent) => void;
}

const MAX_VISIBLE = 3;

export function AllDayRow({ dates, events, onEventClick }: AllDayRowProps) {
  const t = useTranslations("calendar");
  const [expanded, setExpanded] = useState(false);

  // Filter all-day events for each date
  const allDayByDate = dates.map((date) => {
    const dateStr = getLocalDateString(date);
    const dayEvents = events.get(dateStr) || [];
    return dayEvents.filter((e) => e.start_time === null);
  });

  // Check if any column has overflow
  const hasOverflow = allDayByDate.some((items) => items.length > MAX_VISIBLE);

  // If no all-day events at all, don't render the row
  if (allDayByDate.every((items) => items.length === 0)) {
    return null;
  }

  return (
    <div className="border-b border-border bg-background sticky top-0 z-20">
      <div
        className="grid"
        style={{
          gridTemplateColumns: `60px repeat(${dates.length}, 1fr)`,
        }}
      >
        {/* Time gutter label */}
        <div className="flex items-start justify-end pr-2 pt-1 text-xs text-muted-foreground border-r border-border">
          {t("timeGrid.allDay")}
        </div>

        {/* Day columns */}
        {allDayByDate.map((dayAllDayEvents, colIdx) => {
          const visible = expanded
            ? dayAllDayEvents
            : dayAllDayEvents.slice(0, MAX_VISIBLE);
          const remaining = dayAllDayEvents.length - MAX_VISIBLE;

          return (
            <div
              key={colIdx}
              className="px-1 py-1 space-y-0.5 border-r border-border last:border-r-0 min-h-[28px]"
            >
              {visible.map((event) => (
                <div
                  key={event.id}
                  role="button"
                  tabIndex={0}
                  onClick={() => onEventClick?.(event)}
                  onKeyDown={(e) => {
                    if (e.key === "Enter" || e.key === " ") {
                      onEventClick?.(event);
                    }
                  }}
                  className="cursor-pointer"
                >
                  <EventChip event={event} />
                </div>
              ))}
              {!expanded && remaining > 0 && (
                <button
                  type="button"
                  className="text-xs text-muted-foreground hover:text-foreground cursor-pointer"
                  onClick={() => setExpanded(true)}
                >
                  {t("timeGrid.allDayMore", { count: remaining })}
                </button>
              )}
            </div>
          );
        })}
      </div>

      {/* Collapse button when expanded */}
      {expanded && hasOverflow && (
        <button
          type="button"
          className="w-full text-xs text-muted-foreground hover:text-foreground py-0.5 text-center cursor-pointer"
          onClick={() => setExpanded(false)}
        >
          {t("timeGrid.collapseAllDay")}
        </button>
      )}
    </div>
  );
}
```

```tsx
"use client";

import { useState, useEffect } from "react";

const HOUR_HEIGHT = 48; // pixels per hour, matching TimeGrid

/**
 * Teal horizontal line with circle dot indicating current time.
 * Updates position every minute. Only rendered on today's column.
 */
export function CurrentTimeIndicator() {
  const [position, setPosition] = useState(() => computePosition());

  useEffect(() => {
    // Update every 60 seconds
    const interval = setInterval(() => {
      setPosition(computePosition());
    }, 60000);

    return () => clearInterval(interval);
  }, []);

  return (
    <div
      className="absolute left-0 right-0 z-10 pointer-events-none"
      style={{ top: `${position}px` }}
      aria-hidden="true"
    >
      {/* Circle dot on the left edge */}
      <div
        className="absolute -left-1.5 -top-1.5 h-3 w-3 rounded-full bg-[hsl(var(--calendar-event))]"
      />
      {/* Horizontal line */}
      <div className="h-[2px] w-full bg-[hsl(var(--calendar-event))]" />
    </div>
  );
}

function computePosition(): number {
  const now = new Date();
  const hours = now.getHours();
  const minutes = now.getMinutes();
  return (hours + minutes / 60) * HOUR_HEIGHT;
}
```
