<action>
Create `components/calendar/use-calendar-actions.ts`:

A custom hook that returns handler functions for each domain's inline action and a unified `handleItemAction` dispatcher.

```ts
"use client";

import { useCallback } from "react";
import { useRouter } from "next/navigation";
import { useSWRConfig } from "swr";
import type { CalendarItem } from "@/lib/calendar/types";

interface UseCalendarActionsOptions {
  feedKey: string;  // SWR key for the feed to mutate
}

export function useCalendarActions({ feedKey }: UseCalendarActionsOptions) {
  const router = useRouter();
  const { mutate } = useSWRConfig();

  const toggleTask = useCallback(async (item: CalendarItem) => {
    const taskId = item.meta.task_id || item.id;
    const newCompleted = !item.meta.is_completed;

    // Optimistic update
    mutate(feedKey, (current: { items: CalendarItem[] } | undefined) => {
      if (!current) return current;
      return {
        items: current.items.map(i =>
          i.id === item.id
            ? { ...i, meta: { ...i.meta, is_completed: newCompleted } }
            : i
        ),
      };
    }, { revalidate: false });

    try {
      await fetch(`/api/tasks/${taskId}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ is_completed: newCompleted }),
      });
      mutate(feedKey); // Revalidate
    } catch {
      mutate(feedKey); // Rollback by revalidating
    }
  }, [feedKey, mutate]);

  const toggleHabit = useCallback(async (item: CalendarItem) => {
    const habitId = item.meta.habit_id;
    const date = item.meta.habit_log_date;
    if (!habitId || !date) return;

    const wasLogged = item.meta.is_logged;

    // Optimistic update
    mutate(feedKey, (current: { items: CalendarItem[] } | undefined) => {
      if (!current) return current;
      return {
        items: current.items.map(i =>
          i.id === item.id
            ? { ...i, meta: { ...i.meta, is_logged: !wasLogged } }
            : i
        ),
      };
    }, { revalidate: false });

    try {
      if (wasLogged) {
        await fetch(`/api/habits/${habitId}/logs?date=${date}`, {
          method: "DELETE",
        });
      } else {
        await fetch(`/api/habits/${habitId}/logs`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ date, completed: true }),
        });
      }
      mutate(feedKey);
    } catch {
      mutate(feedKey);
    }
  }, [feedKey, mutate]);

  const markBillPaid = useCallback(async (item: CalendarItem) => {
    const billId = item.meta.bill_id || item.id;

    // Optimistic update
    mutate(feedKey, (current: { items: CalendarItem[] } | undefined) => {
      if (!current) return current;
      return {
        items: current.items.map(i =>
          i.id === item.id
            ? { ...i, meta: { ...i.meta, is_paid: true } }
            : i
        ),
      };
    }, { revalidate: false });

    try {
      await fetch(`/api/money/bills/${billId}`, {
        method: "PATCH",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ user_status: "confirmed" }),
      });
      mutate(feedKey);
    } catch {
      mutate(feedKey);
    }
  }, [feedKey, mutate]);

  const navigateToWorkout = useCallback((item: CalendarItem) => {
    const workoutId = item.meta.workout_id || item.id;
    router.push(`/workouts/${workoutId}`);
  }, [router]);

  // Unified dispatcher
  const handleItemAction = useCallback((item: CalendarItem) => {
    switch (item.source) {
      case "task": return toggleTask(item);
      case "habit": return toggleHabit(item);
      case "bill": return markBillPaid(item);
      case "workout": return navigateToWorkout(item);
      default: break;
    }
  }, [toggleTask, toggleHabit, markBillPaid, navigateToWorkout]);

  return { handleItemAction, toggleTask, toggleHabit, markBillPaid, navigateToWorkout };
}
```
</action>
