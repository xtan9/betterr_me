/**
 * Get tasks with due_date within a date range (inclusive).
 * Used by the calendar feed API to show tasks on the calendar.
 */
async getTasksInDateRange(userId: string, startDate: string, endDate: string): Promise<Task[]> {
  const { data, error } = await this.supabase
    .from('tasks')
    .select('*')
    .eq('user_id', userId)
    .not('due_date', 'is', null)
    .gte('due_date', startDate)
    .lte('due_date', endDate)
    .order('due_date', { ascending: true });

  if (error) throw error;
  return data || [];
}
```

This method filters tasks that have a `due_date` set and fall within the specified date range. It returns all tasks (both completed and incomplete) so the calendar can show completion status.

/**
 * Get completed/discarded workouts within a date range.
 * Filters by started_at timestamp, converting date strings to timestamp bounds.
 * Used by the calendar feed API.
 */
async getWorkoutsByDateRange(
  userId: string,
  startDate: string,
  endDate: string
): Promise<Workout[]> {
  const { data, error } = await this.supabase
    .from("workouts")
    .select("*")
    .eq("user_id", userId)
    .neq("status", "in_progress")
    .gte("started_at", `${startDate}T00:00:00`)
    .lte("started_at", `${endDate}T23:59:59`)
    .order("started_at", { ascending: true });

  if (error) {
    log.error("Failed to get workouts by date range", error);
    throw error;
  }

  return data ?? [];
}
```

Create `lib/calendar/normalize.ts` with 5 normalize functions that convert domain records to `CalendarItem[]`:

1. **`normalizeEvents(events: ExpandedCalendarEvent[]): CalendarItem[]`**
   - Map each event: `source: "event"`, `color: event.color || DOMAIN_COLORS.event`
   - `is_all_day: event.start_time === null`
   - `meta: { location: event.location, description: event.description, is_virtual: event.is_virtual, recurrence_rule: event.recurrence_rule }`

2. **`normalizeTasks(tasks: Task[]): CalendarItem[]`**
   - Filter to only tasks with `due_date` (should already be filtered, but defensive)
   - `source: "task"`, `color: DOMAIN_COLORS.task`
   - `start_date: task.due_date`, `start_time: task.due_time`, `end_date: task.due_date`, `end_time: null`
   - `is_all_day: task.due_time === null`
   - `meta: { is_completed: task.is_completed, priority: task.priority, task_id: task.id }`

3. **`normalizeHabits(habits: Habit[], logs: HabitLog[], startDate: string, endDate: string): CalendarItem[]`**
   - Iterate each date in [startDate, endDate] range
   - For each date, check each habit with `shouldTrackOnDate(habit.frequency, dateObj)`
   - Build a `Set` of `"habitId:date"` keys from logs for O(1) lookup
   - `id: "habit-{habitId}-{date}"`, `source: "habit"`, `color: DOMAIN_COLORS.habit`
   - `start_date: date`, `start_time: null`, `is_all_day: true`
   - `meta: { is_logged: logSet.has(key), habit_id: habit.id, habit_log_date: date }`

4. **`normalizeBills(bills: RecurringBill[], startDate: string, endDate: string): CalendarItem[]`**
   - Filter bills: `is_active === true`, `user_status !== 'dismissed'`, `next_due_date` within range
   - `source: "bill"`, `color: DOMAIN_COLORS.bill`
   - `start_date: bill.next_due_date`, `start_time: null`, `is_all_day: true`
   - `meta: { is_paid: bill.user_status === 'confirmed', amount_cents: bill.amount_cents, bill_id: bill.id }`

5. **`normalizeWorkouts(workouts: Workout[], timezone?: string): CalendarItem[]`**
   - Accept an optional `timezone` parameter (IANA timezone string, e.g. `"America/New_York"`). The caller (feed API) passes the user's timezone from the request query params or defaults to UTC.
   - Extract date and time from `started_at` using timezone-aware conversion: use `new Date(started_at).toLocaleString("sv-SE", { timeZone })` to get local date/time parts (project convention: dates are always browser-local, never UTC). If no timezone provided, fall back to splitting the ISO string directly (existing behavior).
   - `source: "workout"`, `color: DOMAIN_COLORS.workout`
   - `start_time`: extracted local time portion, `is_all_day: false`
   - `end_time`: if `duration_seconds` available, calculate end time
   - `meta: { workout_id: workout.id, duration_seconds: workout.duration_seconds }`

Import `shouldTrackOnDate` from `@/lib/habits/format`.
