    const handleSend = useCallback(async (text: string) => {
      let convId = activeConversationId;

      // D-11: Auto-create conversation on first message if none selected
      if (!convId) {
        try {
          const res = await fetch("/api/conversations", { method: "POST" });
          const data = await res.json();
          convId = data.conversation.id;
          setActiveConversationId(convId);
          // Update URL without full navigation
          window.history.replaceState(null, "", `/chat?id=${convId}`);
          mutateConversations(); // Refresh sidebar
        } catch (err) {
          console.error("[chat] Failed to create conversation:", err);
          return;
        }
      }

      // D-04: Save user message to DB BEFORE sending to LLM
      try {
        await fetch(`/api/conversations/${convId}/messages`, {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({ role: "user", content: text }),
        });
      } catch (err) {
        console.error("[chat] Failed to save user message:", err);
      }

      // Send to LLM
      sendMessage({ text });
    }, [activeConversationId, sendMessage, mutateConversations]);

    **onFinish callback for assistant message persistence (per D-05) and title generation (per D-07/D-08):**

    Pass `onFinish` to `useChat`. Note: `useChat` from `@ai-sdk/react` accepts `onFinish` in the hook options. Check the actual API — it may be on the hook config or need to be handled differently. The executor should read the `@ai-sdk/react` types to confirm.

    The approach: use a `useEffect` that watches `messages` and `status`. When status transitions from "streaming" to "ready" and the last message is from "assistant", save the assistant message and trigger title if it's the first exchange.

    ```typescript
    const prevStatusRef = useRef(status);
    useEffect(() => {
      const wasStreaming = prevStatusRef.current === "streaming" || prevStatusRef.current === "submitted";
      prevStatusRef.current = status;

      if (wasStreaming && status === "ready" && messages.length > 0) {
        const lastMsg = messages[messages.length - 1];
        if (lastMsg.role === "assistant" && activeConversationId) {
          const textPart = lastMsg.parts.find(p => p.type === "text");
          const content = textPart?.type === "text" ? textPart.text : "";

          // D-05: Save assistant message after stream completes
          fetch(`/api/conversations/${activeConversationId}/messages`, {
            method: "POST",
            headers: { "Content-Type": "application/json" },
            body: JSON.stringify({ role: "assistant", content }),
          }).catch(err => console.error("[chat] Failed to save assistant message:", err));

          // D-07/D-08: Auto-generate title after first exchange (fire-and-forget)
          // First exchange = exactly 2 messages (1 user + 1 assistant)
          if (messages.length === 2) {
            const userMsg = messages[0];
            const userText = userMsg.parts.find(p => p.type === "text");
            fetch(`/api/conversations/${activeConversationId}/title`, {
              method: "POST",
              headers: { "Content-Type": "application/json" },
              body: JSON.stringify({
                userMessage: userText?.type === "text" ? userText.text : "",
                assistantMessage: content,
              }),
            })
              .then(() => mutateConversations()) // Refresh sidebar to show new title
              .catch(err => console.error("[chat] Failed to generate title:", err));
          }

          // Refresh conversation list to update updated_at ordering
          mutateConversations();
        }
      }
    }, [status, messages, activeConversationId, mutateConversations]);
    ```
